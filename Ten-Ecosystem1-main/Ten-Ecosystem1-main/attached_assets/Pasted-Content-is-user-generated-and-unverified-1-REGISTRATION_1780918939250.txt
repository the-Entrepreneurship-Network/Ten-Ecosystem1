// Universal question count based on week number
function getQuestionCountForWeek(weekNumber) {
  if (weekNumber <= 3) return 10      // 15 min timer
  if (weekNumber <= 6) return 15      // 20 min timer
  if (weekNumber <= 9) return 20      // 25 min timer
  if (weekNumber <= 12) return 25     // 30 min timer
  return 30                           // 40 min timer (6-month extended)
}

function getTimerForWeek(weekNumber) {
  if (weekNumber <= 3) return 15 * 60   // 15 minutes in seconds
  if (weekNumber <= 6) return 20 * 60
  if (weekNumber <= 9) return 25 * 60
  if (weekNumber <= 12) return 30 * 60
  return 40 * 60
}

function getPassScoreForWeek(weekNumber, totalQuestions) {
  let passPct = 0.60  // default 60%
  if (weekNumber > 3) passPct = 0.65
  if (weekNumber > 6) passPct = 0.70
  if (weekNumber > 9) passPct = 0.75
  if (weekNumber > 12) passPct = 0.80
  return Math.ceil(totalQuestions * passPct)
}
