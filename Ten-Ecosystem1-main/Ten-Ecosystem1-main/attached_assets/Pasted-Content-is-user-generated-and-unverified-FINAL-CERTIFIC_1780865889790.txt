javascript
// NEW FEATURE: Load fellowship card state on page load
async function loadFellowshipCard() {
  const token = getToken()
  try {
    const response = await fetch('/api/v2/certificates/my-certs', {
      headers: { 'Authorization': `Bearer ${token}` }
    })
    const data = await response.json()

    const fellowshipData = data.fellowship

    if (fellowshipData.eligible) {
      // Show full fellowship card
      document.getElementById('fellowship-card').style.display = 'none'
      document.getElementById('fellowship-card-eligible').style.display = 'block'

      if (fellowshipData.cohort_rank_percent !== undefined) {
        document.getElementById('fellowship-rank-display').textContent =
          `Top ${Math.round(fellowshipData.cohort_rank_percent)}%`
      }
    } else {
      // Show mystery card
      document.getElementById('fellowship-card').style.display = 'block'
      document.getElementById('fellowship-card-eligible').style.display = 'none'

      // Show cohort rank in mystery card
      if (fellowshipData.cohort_rank_percent !== undefined) {
        document.getElementById('cohort-rank-display').textContent =
          `${Math.round(fellowshipData.cohort_rank_percent)}% rank`
      }
    }

    // Handle issued state
    if (fellowshipData.status === 'issued') {
      showIssuedState('fellowship', fellowshipData)
    }

  } catch (error) {
    console.error('Failed to load fellowship card:', error)
  }
}

// Call on page load
document.addEventListener('DOMContentLoaded', () => {
  loadCertificates()
  loadFellowshipCard()
})
