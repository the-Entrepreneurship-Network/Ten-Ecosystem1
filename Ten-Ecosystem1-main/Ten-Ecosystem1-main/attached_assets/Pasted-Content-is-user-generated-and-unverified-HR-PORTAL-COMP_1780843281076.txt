function generateDocumentNumber(type) {
  const prefixes = {
    'offer_letter': 'TEN-OL',
    'lor': 'TEN-LOR',
    'expert': 'TEN-EXP',
    'nano_degree': 'TEN-ND',
    'fellowship': 'TEN-FEL'
  }
  const year = new Date().getFullYear()
  const random = Math.random().toString(36).substring(2,8).toUpperCase()
  return `${prefixes[type] || 'TEN-DOC'}-${year}-${random}`
}

// scripts/backfillDocumentHistory.js
async function backfill() {
  const records = await DocumentHistory.find({
    $or: [
      { student_name: { $in: [null, '', undefined, '—'] } },
      { document_number: { $in: [null, '', undefined] } }
    ]
  })
  for (const record of records) {
    const student = await User.findById(record.student_id)
    if (student) {
      record.student_name = student.name || student.email
      record.college = student.college || 'Not provided'
    }
    if (!record.document_number) {
      record.document_number = generateDocumentNumber(
        record.document_type || 'offer_letter'
      )
    }
    await record.save()
  }
  console.log(`Fixed ${records.length} records`)
}

// Route: POST /api/hr/verify-document
// or GET /api/v2/verify/:documentNumber (public)

async function verifyDocument(documentNumber) {
  if (!documentNumber) {
    return { verified: false, message: 'Please enter a document number' }
  }

  const cleaned = documentNumber.trim().toUpperCase()

  const record = await DocumentHistory.findOne({
    document_number: cleaned
  })

  if (!record) {
    return {
      verified: false,
      message: 'Document not found. Check the number and try again.'
    }
  }

  return {
    verified: true,
    document_number: record.document_number,
    student_name: record.student_name,
    employee_id: record.employee_id,
    document_type: record.document_type,
    domain: record.domain || 'N/A',
    college: record.college || 'N/A',
    issued_date: record.sent_on || record.createdAt,
    issued_by: 'The Entrepreneurship Network (TEN)'
  }
}
