// NEW FEATURE: Fellowship Eligibility Check
// This function must be called EVERY TIME the certificates page loads

async function checkFellowshipEligibility(studentId) {
  try {
    // Step 1: Get student's course completion percentage
    const totalTasks = await StudentTaskProgress.countDocuments({
      student_id: studentId
    })
    const completedTasks = await StudentTaskProgress.countDocuments({
      student_id: studentId,
      status: 'approved'
    })
    const completionPercent = totalTasks > 0
      ? (completedTasks / totalTasks) * 100
      : 0

    // Step 2: Get student's cohort rank
    const student = await User.findById(studentId)
    const cohortId = student.cohort_id

    // Count total students in same cohort
    const totalCohortStudents = await User.countDocuments({
      cohort_id: cohortId,
      role: 'student'
    })

    // Get student's coin rank in cohort
    const studentCoins = await StudentCoin.findOne({ student_id: studentId })
    const studentTotalCoins = studentCoins?.total_coins || 0

    // Count how many cohort students have MORE coins than this student
    const studentsAhead = await StudentCoin.countDocuments({
      student_id: {
        $in: await User.find({ cohort_id: cohortId, role: 'student' })
          .distinct('_id')
      },
      total_coins: { $gt: studentTotalCoins }
    })

    // Calculate percentile rank (lower number = better rank)
    const cohortRankPercent = totalCohortStudents > 0
      ? (studentsAhead / totalCohortStudents) * 100
      : 100

    // Step 3: Apply strict eligibility rules
    const isEligible = (
      completionPercent >= 70 &&        // Must be 70%+ complete
      cohortRankPercent <= 10            // Must be in top 10%
    )

    return {
      isEligible,
      completionPercent: Math.round(completionPercent),
      cohortRankPercent: Math.round(cohortRankPercent),
      reason: !isEligible
        ? `Needs ${Math.max(0, 70 - completionPercent).toFixed(0)}% more completion
           and top 10% cohort rank (currently ${cohortRankPercent.toFixed(0)}%)`
        : 'Eligible'
    }
  } catch (error) {
    console.error('Fellowship eligibility check error:', error)
    return { isEligible: false, completionPercent: 0, cohortRankPercent: 100 }
  }
}

// NEW FEATURE: LOC Auto-Generation
// Trigger: when student completes 100% of tasks

async function autoGenerateLOC(studentId) {
  const totalTasks = await StudentTaskProgress.countDocuments({
    student_id: studentId
  })
  const completedTasks = await StudentTaskProgress.countDocuments({
    student_id: studentId,
    status: 'approved'
  })

  if (totalTasks > 0 && completedTasks === totalTasks) {
    // Student completed 100% — generate LOC
    const student = await User.findById(studentId)

    const locExists = await DocumentHistory.findOne({
      student_id: studentId,
      document_type: 'loc'
    })

    if (!locExists) {
      const docNumber = generateDocumentNumber('loc')
      // Generate LOC PDF using same Puppeteer flow as offer letter
      const locPdf = await generateLOCPdf(student, docNumber)
      const locUrl = await saveToStorage(locPdf, `loc_${studentId}.pdf`)

      await DocumentHistory.create({
        student_id: studentId,
        student_name: student.name,
        document_type: 'loc',
        document_number: docNumber,
        pdf_url: locUrl,
        sent_on: new Date(),
        sent_to_email: student.email
      })

      // Email LOC to student
      await sendEmail(student.email, 'Your Letter of Completion', locUrl)

      console.log(`LOC generated for student: ${student.name}`)
    }
  }
}

// NEW FEATURE: Psychology Trigger Checker
// Runs every hour via cron

const cron = require('node-cron')

cron.schedule('0 * * * *', async () => {
  console.log('[CRON] Checking psychology triggers...')

  const students = await User.find({ role: 'student', isActive: true })

  for (const student of students) {
    const daysSinceRegistration = Math.floor(
      (Date.now() - new Date(student.createdAt)) / (1000 * 60 * 60 * 24)
    )
    const completionPercent = await getStudentCompletionPercent(student._id)
    const duration = student.duration || '3months'

    // Check Expert Cert trigger timing
    const expertThresholds = {
      '45days': 10, '1month': 7, '3months': 42, '6months': 60
    }
    const expertThreshold = expertThresholds[duration]

    if (
      daysSinceRegistration >= expertThreshold &&
      completionPercent >= 30
    ) {
      await logTriggerIfNotShown(student._id, 'EXPERT_CERT_UNLOCK')
    }

    // Check Nano Degree trigger
    const nanoThresholds = {
      '45days': 30, '1month': 22, '3months': 84, '6months': 120
    }
    const nanoThreshold = nanoThresholds[duration]

    if (
      daysSinceRegistration >= nanoThreshold &&
      completionPercent >= 70
    ) {
      await logTriggerIfNotShown(student._id, 'NANO_CERT_UNLOCK')
    }

    // Check Fellowship eligibility (strict)
    const fellowship = await checkFellowshipEligibility(student._id)
    if (fellowship.isEligible) {
      await logTriggerIfNotShown(student._id, 'FELLOWSHIP_WHISPER')
    }
  }
})

async function logTriggerIfNotShown(studentId, triggerName) {
