    const score = (stats.attendance * 0.3) + (stats.performance * 0.3)
                + (stats.taskSubmission * 0.25) + Math.min(stats.streak, 100) * 0.15;
    if (score >= 90) return { grade: '⭐⭐⭐⭐⭐', label: 'Outstanding Performer' };

  }
